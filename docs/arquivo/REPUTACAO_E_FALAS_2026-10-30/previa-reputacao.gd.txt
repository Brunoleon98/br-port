extends "res://tools/capturar_tela.gd"

func _initialize() -> void:
	if OS.get_environment("BRPORT_FOTO_TAMANHO") == "maior":
		PreferenciasDeApresentacao.atual().tamanho_do_texto = PreferenciasDeApresentacao.MAIOR

func _montar() -> void:
	super()
	var valor := float(OS.get_environment("BRPORT_FOTO_REPUTACAO"))
	GS._change_reputation(valor - GS.reputation)
	var rotulo: Label = _main.get_node("HudBar/RepPilula/Linha/RepTexto")
	# Prévia da única linha proposta para o Main. O Main de produção não foi editado.
	AvisoDeReputacao.atualizar(rotulo, GS.reputation)
	var motor = load("res://scripts/validation/contraste_ui.gd").new()
	var fundo: Dictionary = motor.fundo_de(rotulo, _main)
	print("PREVIA: ", rotulo.text, " | ", rotulo.theme_type_variation)
	print("FUNDO: ", fundo)
	for cor in fundo["cores"]:
		print("CONTRASTE: ", motor.contraste(rotulo.get_theme_color("font_color"), cor))

func _process(delta: float) -> bool:
	if _montado and is_instance_valid(_main):
		GS.reputation = float(OS.get_environment("BRPORT_FOTO_REPUTACAO"))
		_main._refresh_hud()
		AvisoDeReputacao.atualizar(_main.get_node("HudBar/RepPilula/Linha/RepTexto"), GS.reputation)
	return super(delta)
